Stamp from the command line

The API is the primary interface. The browser is a convenience.

curl — open tier, no auth
# Compute locally, submit only the digest
DIGEST=$(sha256sum ./contract-v4.pdf | cut -d' ' -f1)

curl -sS https://api.sanctumsecops.org/log/append \
  -H 'content-type: application/json' \
  -d "{\"digest\":\"sha256:$DIGEST\"}"

# => { "leaf": 1284417, "pending_window": "2026-08-27T18:05:00Z" }

# Collect the receipt once the window closes and the tree head is signed
curl -sS https://api.sanctumsecops.org/log/inclusion-proof?digest=sha256:$DIGEST \
  -o contract-v4.pdf.ots
